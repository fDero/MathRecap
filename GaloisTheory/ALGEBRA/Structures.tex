\subsection{Algebraic Structures}
In this chapter we will define the algebraic structures that we will be using in the 
rest of the note. In particular, we will define Monoids, Groups, Rings and Fields.  

\begin{definition}[Monoid]
    A monoid $(M, \perp)$ is a set $M$ equipped with an internal binary operation 
    $\perp$ that is both associative, and has some identity element $\epsilon$.
\end{definition}

\begin{definition}[Group, Abelian Group]
    A group $(G, \perp)$ is monoid for which every element has an inverse. A 
    group is abelian if the operation $\perp$ is commutative.
\end{definition}

\begin{lemma}
    If $(G, \perp)$ is a group, then, for every element $a \in A$, the identity
    $k \perp k = k$ implies that $a$ is the identity element.
\end{lemma}
\begin{proof}
    $$ a \perp \epsilon = a $$
    $$ a \perp (k \perp k^{-1}) = a $$
    $$ a \perp (k \perp k \perp k^{-1}) = a $$
    $$ a \perp k \perp (k \perp k^{-1}) = a $$
    $$ a \perp k \perp \epsilon = a $$
    $$ a \perp k = a $$
\end{proof}

\begin{lemma}[Cancellation-law]
    If $(G, \perp)$ is a group, then, $f(x) = x \perp k$ is injective.
\end{lemma}
\begin{proof}
    $$ x_1 \perp k = x_2 \perp k $$
    $$ x_1 \perp k \perp k^{-1} = x_2 \perp k \perp k^{-1} $$
    $$ x_1 = x_2 $$
\end{proof}

\begin{definition}[Ring]
    A ring $(A, \perp, \nabla)$ is a set $A$ equipped with two internal binary operations
    $\perp$ and $\nabla$ such that $(A, \perp)$ is an abelian group, $(A, \nabla)$ is a monoid, 
    and the two operations are compatible in the sense that for all $a, b, c \in A$:
    \begin{align*}
        a \nabla (b \perp c) &= (a \nabla b) \perp (a \nabla c)\\
        (a \perp b) \nabla c &= (a \nabla c) \perp (b \nabla c)
    \end{align*}
\end{definition} \vspace{1em}

\begin{lemma}[Absorbing element]
    If $(A, \perp, \nabla)$ is a ring, then the identity element of $(A, \perp)$ 
    is the absorbing element of $(A, \nabla)$. Meaning that an absorbing element
    always exists.
\end{lemma}
\begin{proof}
    Assuming $\theta$ to be the identity element of $(A, \perp)$, we have that the following 
    holds:
    $$ a \nabla \theta = a \nabla (\theta \perp \theta) = (a \nabla \theta) \perp (a \nabla \theta) $$
\end{proof}

\begin{definition}[Field]
    A field $(F, \perp, \nabla)$ is a field for which every element has an inverse 
    with respect to $\nabla$ except the absorbing element, meaning that $(F - \{\theta\}, \nabla)$
    is a group.
\end{definition}

\begin{example}
    An example of commutative monoid is $(\mathbb{N}, +)$, where the $+$ operation is the 
    classical sum of natural numbers. The identity element is $0$, and the operation is 
    associative. There are no inverse elements for non-zero numbers such that 
    their sum is $0$. \\
\end{example}

\begin{example}
    Another example of commutative monoid is $(P(S), \cup)$, where $P(S)$ is the power set of a 
    set $S$,and the $\cup$ operation is the union of sets. The identity element is the empty set, 
    and the operation is associative. There are no inverses. \\
\end{example}

\begin{example}
    An example of non-commutative monoid is $(\{\;f : S \rightarrow S \;\}, \circ)$, where the 
    $\circ$ operation is the function composition. The identity element is the identity function. 
    Not all functions are invertible. \\
\end{example}

\begin{example}
    Another example of commutative monoid is $(P(S), \cap)$, where $P(S)$ is the power set of a 
    set $S$,and the $\cap$ operation is the intersection of sets. The identity element is the 
    set $S$, and the operation is associative. There are no inverses. \\
\end{example}

\begin{example}
    An example of non-abelian group is $(\mathbb{R}_{2,2}, \times)$, where $\mathbb{R}_{2,2}$ 
    is the set of two by two matrices with real coefficients, and the $\times$ operation is the 
    matrix product. The operation is associative, there is an identity element, and the
    inverse of a matrix always exists. However, the operation is not commutative. \\ 
\end{example}

\begin{example}
    An example of non-abelian group is the group $(B(S), \circ)$ of all bijective functions from 
    a set to itself under the operation of function composition. The inverse is the identity 
    function and the operation is associative. Bijections are invertible. \\ 
\end{example}

\begin{example}
    An example of abelian group is $(\mathbb{Q}, \cdot)$, where the $\cdot$ operation is the 
    classical multiplication of rational numbers. The identity element is $1$, and the 
    operation is associative. The inverse of $x$ is $\frac{1}{x}$. \\
\end{example}

\begin{example}
    An example of ring is $(\mathbb{Z}, +, \cdot)$, where the $+$ operation is the classical sum of
    integers, and the $\cdot$ operation is the classical multiplication of integers. \\
\end{example}

\begin{example}
    Another ring is $(\mathbb{R}[x], +, \cdot)$ where $\mathbb{R}[x]$ is the set of
    polynomials with real coefficients, and the $+$ and $\cdot$ operations are the classical sum and
    of polynomials. The reason why this is not a field is that the polynomial $x$ does not have a
    multiplicative inverse such that $x \cdot f(x) = 1$. In general, $F[x]$ where $F$ is a field, 
    forms a ring. \\
\end{example}

\begin{example}
    Canonical examples of fields are $(\mathbb{Q}, +, \cdot)$, $(\mathbb{R}, +, \cdot)$, 
    $(\mathbb{C}, +, \cdot)$. \\
\end{example}
